SELECT 
    p.project_name,
    m.member_id,
    m.full_name,
    -- Each week as a separate column using the parameterized week values
    SUM(CASE WHEN t.week_id = {{SelectedWeek.value}} THEN t.actual_hours ELSE 0 END) AS week_1,
    SUM(CASE WHEN t.week_id = {{SelectedWeek.value+1}} THEN t.actual_hours ELSE 0 END) AS week_2,
    SUM(CASE WHEN t.week_id = {{SelectedWeek.value+2}} THEN t.actual_hours ELSE 0 END) AS week_3,
    SUM(CASE WHEN t.week_id = {{SelectedWeek.value+3}} THEN t.actual_hours ELSE 0 END) AS week_4,
    SUM(CASE WHEN t.week_id = {{SelectedWeek.value+4}} THEN t.actual_hours ELSE 0 END) AS week_5,
		SUM(CASE WHEN t.week_id BETWEEN {{SelectedWeek.value}} AND {{SelectedWeek.value + 4}} THEN t.actual_hours ELSE 0 END) AS total_efforts
FROM 
    public.timesheet t
JOIN 
    public."member" m ON t.member_id = m.member_id
JOIN 
    public.project p ON t.project_id = p.project_id
WHERE 
    EXTRACT(YEAR FROM t.working_date) = {{SelectedYear.value}}
GROUP BY 
    p.project_name, m.member_id, m.full_name
ORDER BY 
    p.project_name, m.member_id;
