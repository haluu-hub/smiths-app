DROP FUNCTION IF EXISTS public.get_project_weekly_effort(integer, date, date);

CREATE OR REPLACE FUNCTION public.get_project_weekly_effort(week_no integer, start_date date, end_date date)
 RETURNS TABLE(
    project character varying, 
    pm character varying, 
    department character varying, 
    p integer, 
    pl1 numeric, 
    at1 numeric, 
    pl2 numeric, 
    at2 numeric, 
    pl3 numeric, 
    at3 numeric, 
    pl4 numeric, 
    at4 numeric, 
    pl5 numeric, 
    at5 numeric, 
    tpl numeric, 
    tat numeric,
    own_project_effort numeric,
    support_project_effort numeric
  )
 LANGUAGE plpgsql
AS $function$
BEGIN
    RETURN QUERY
    
        select pp.project_name, pp.project_manager, d.department_name,  aa.* from (WITH weekly_allocations AS (
        SELECT 
            project_id,
            ((allocation_date - DATE '1970-01-01' + 3) / 7) as week_id,
            SUM(allocation_hours) AS wah
        FROM 
            public.project_member_daily_allocation
        WHERE 
            allocation_date BETWEEN $2 AND $3
        GROUP BY 
            project_id, week_id
    ),
     weekly_actuals as (SELECT 
            project_id, week_id, 
            SUM(actual_hours) as ah,
            SUM(own_actual_hours) as own_ah,
            SUM(support_actual_hours) as support_ah
        from (SELECT 
            project_id,
            ((working_date - DATE '1970-01-01' + 3) / 7) as week_id,
            SUM(actual_hours) actual_hours,
            SUM(CASE WHEN is_support = false OR is_support IS NULL THEN actual_hours ELSE 0 END) as own_actual_hours,
            SUM(CASE WHEN is_support = true THEN actual_hours ELSE 0 END) as support_actual_hours
        FROM 
            public.timesheet
        WHERE 
            working_date BETWEEN $2 AND $3
        GROUP BY 
            project_id, working_date) as CCC
        GROUP by
            project_id, week_id)
            SELECT 
        coalesce(w.project_id, wa.project_id)  p,
        COALESCE(SUM(CASE WHEN w.week_id = $1 then w.wah END), 0) AS pl1,
        COALESCE(SUM(CASE WHEN wa.week_id = $1 then wa.ah END), 0) AS at1,
        COALESCE(SUM(CASE WHEN w.week_id = $1 + 1 then w.wah END), 0) AS pl2,
        COALESCE(SUM(CASE WHEN wa.week_id = $1+1 then wa.ah END), 0) AS at2,
        COALESCE(SUM(CASE WHEN w.week_id = $1+2 then w.wah END), 0) AS pl3,
        COALESCE(SUM(CASE WHEN wa.week_id = $1+2 then wa.ah END), 0) AS at3,
        COALESCE(SUM(CASE WHEN w.week_id = $1+3 then w.wah END), 0) AS pl4,
        COALESCE(SUM(CASE WHEN wa.week_id = $1+3 then wa.ah END), 0) AS at4,
        COALESCE(SUM(CASE WHEN w.week_id = $1+4 then w.wah END), 0) AS pl5,
        COALESCE(SUM(CASE WHEN wa.week_id = $1+4 then wa.ah END), 0) AS at5,
        COALESCE(SUM(w.wah), 0) AS tpl, 
        COALESCE(SUM(wa.ah),0) AS tat,
        COALESCE(SUM(wa.own_ah),0) AS own_project_effort,
        COALESCE(SUM(wa.support_ah),0) AS support_project_effort
        
    FROM 
        weekly_allocations w 
        full outer join weekly_actuals wa on w.project_id = wa.project_id and w.week_id = wa.week_id
    GROUP BY 
        p)  aa
        left join project pp on aa.p = pp.project_id
        left join department d on d.department_id = pp.department_id;
        
END;
$function$;
